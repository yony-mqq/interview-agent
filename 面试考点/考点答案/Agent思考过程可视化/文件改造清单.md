```typescript
// 新增：SSE 流事件类型
export interface ThoughtEvent {
  type: 'thought';
  content: string;        // 思考内容
  step: number;           // 第几步思考
}

export interface ToolCallEvent {
  type: 'tool_call';
  name: string;           // 工具名称
  input: string;          // 调用参数
}

export interface ToolResultEvent {
  type: 'tool_result';
  name: string;           // 工具名称
  output: string;         // 返回结果
}

export interface ContentEvent {
  type: 'content';
  content: string;        // 最终回答片段
}

export type StreamEvent = ThoughtEvent | ToolCallEvent | ToolResultEvent | ContentEvent;
```

### 3. server/src/routes/chat.ts（核心改造）

**修改内容：重构为 ReAct 多步流程**

```typescript
// 改造后的 chat.ts 核心流程变化：

// 原流程（单步）：标准的 RAG（检索增强生成）工作流
// 检索 → 组装 Prompt → 调用 LLM → 流式输出 content

// 新流程（多步 ReAct）：
// Step 1: 输出 thought 事件（意图分析）
//   → 调用轻量模型分析用户意图
//   → res.write(`data: ${JSON.stringify({type:'thought',...})}\n\n`)
//
// Step 2: 输出 tool_call 事件（调用检索工具）
//   → 调用 retrieveRelevantChunks()
//   → res.write(`data: ${JSON.stringify({type:'tool_call',...})}\n\n`)
//
// Step 3: 输出 tool_result 事件（检索结果）
//   → 拿到 chunks
//   → res.write(`data: ${JSON.stringify({type:'tool_result',...})}\n\n`)
//
// Step 4: 流式输出 content 事件（最终回答）
//   → 组装完整 Prompt（含检索结果）
//   → for await (chunk of streamChat(messages))
//   → 每个 chunk 包装为 {type:'content', content: chunk}
```

**具体修改点：**

| 行号范围 | 修改内容 |
|---|---|
| 导入区 | 新增 `StreamEvent` 类型导入 |
| ~line 90（保存用户消息后） | **Step 1**: 插入意图分析，发送 `thought` 事件 |
| ~line 92（原检索代码前） | **Step 2**: 发送 `tool_call` 事件 |
| ~line 103（检索完成后） | **Step 3**: 发送 `tool_result` 事件 |
| ~line 124（流式生成循环） | **Step 4**: 每个 chunk 包装为 `ContentEvent` |

**修改内容：新增 ThinkingStep 和 StreamEvent 类型**

```typescript
// 新增：思考步骤
export interface ThinkingStep {
  id: string;              // 唯一标识
  type: 'thought' | 'tool_call' | 'tool_result';
  icon: string;            // 显示图标，如 '🤔' '🔍' '✍️'
  title: string;           // 步骤标题
  content: string;         // 详细内容
  timestamp: number;       // 时间戳
}

// 新增：消息可携带思考链
export interface Message {
  id?: number;
  role: 'user' | 'assistant';
  content: string;
  created_at?: string;
  thinkingSteps?: ThinkingStep[];  // ← 新增
}

// 新增：SSE 流事件类型（与后端对齐）
export interface StreamEvent {
  type: 'thought' | 'tool_call' | 'tool_result' | 'content' | 'error';
  content?: string;
  name?: string;
  input?: string;
  output?: string;
  error?: string;
}
```

### 5. client/src/hooks/useChat.ts（核心改造）

**修改内容：解析多类型 SSE 事件**

```typescript
// 新增状态
const [thinkingSteps, setThinkingSteps] = useState<ThinkingStep[]>([]);

// sendMessage 中，解析逻辑改造：
// 原逻辑：只解析 parsed.content
// 新逻辑：根据 parsed.type 分别处理

// 伪代码：
const parsed: StreamEvent = JSON.parse(data);

switch (parsed.type) {
  case 'thought':
    setThinkingSteps(prev => [...prev, {
      id: generateId(),
      type: 'thought',
      icon: '🤔',
      title: '正在分析意图',
      content: parsed.content,
      timestamp: Date.now(),
    }]);
    break;

  case 'tool_call':
    setThinkingSteps(prev => [...prev, {
      id: generateId(),
      type: 'tool_call',
      icon: '🔍',
      title: `调用 ${parsed.name}`,
      content: parsed.input,
      timestamp: Date.now(),
    }]);
    break;

  case 'tool_result':
    setThinkingSteps(prev => [...prev, {
      id: generateId(),
      type: 'tool_result',
      icon: '📄',
      title: `${parsed.name} 返回结果`,
      content: parsed.output,
      timestamp: Date.now(),
    }]);
    break;

  case 'content':
    // 追加到 assistant 消息的 content
    setMessages(prev => {
      const last = prev[prev.length - 1];
      if (!last || last.role !== 'assistant') return prev;
      return [...prev.slice(0, -1), {
        ...last,
        content: last.content + parsed.content,
        thinkingSteps: thinkingSteps,  // 把思考链绑定到消息
      }];
    });
    break;
}
```

**具体修改点：**

| 行号范围 | 修改内容 |

| ~line 12（state 区） | 新增 `thinkingSteps` state |
| ~line 83（解析逻辑） | 把 `if (parsed.content)` 改为 `switch(parsed.type)` |
| ~line 110（错误处理） | 清空 thinkingSteps |
| return 对象 | 新增 `thinkingSteps` |

### 6. client/src/components/Chat/MessageBubble.tsx（核心改造）

**修改内容：增加折叠面板展示思考链**

```tsx
// 新增 props
interface MessageBubbleProps {
  message: Message;
  isStreaming?: boolean;
}

// 新增：思考步骤折叠面板组件（可内联或拆分为 ThinkingPanel.tsx）
function ThinkingPanel({ steps }: { steps: ThinkingStep[] }) {
  const [expanded, setExpanded] = useState(true); // 流式时默认展开，完成后可折叠

  if (steps.length === 0) return null;

  return (
    <div className="tw-mb-2 tw-border-b tw-border-gray-200/60 tw-pb-2">
      <button
        onClick={() => setExpanded(!expanded)}
        className="tw-flex tw-items-center tw-gap-1.5 tw-text-xs tw-text-gray-500 hover:tw-text-gray-700 tw-transition"
      >
        <span>{expanded ? '▼' : '▶'}</span>
        <span>🤔 思考过程（{steps.length} 步）</span>
      </button>

      {expanded && (
        <div className="tw-mt-1.5 tw-space-y-1.5">
          {steps.map((step) => (
            <div
              key={step.id}
              className={`tw-flex tw-items-start tw-gap-2 tw-text-xs tw-rounded-lg tw-px-2.5 tw-py-1.5 ${
                step.type === 'thought'
                  ? 'tw-bg-blue-50 tw-text-blue-700'
                  : step.type === 'tool_call'
                  ? 'tw-bg-amber-50 tw-text-amber-700'
                  : 'tw-bg-green-50 tw-text-green-700'
              }`}
            >
              <span className="tw-flex-shrink-0">{step.icon}</span>
              <div className="tw-min-w-0">
                <div className="tw-font-medium">{step.title}</div>
                <div className="tw-text-gray-600 tw-truncate">{step.content}</div>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

// MessageBubble 主体改造：
export function MessageBubble({ message, isStreaming }: MessageBubbleProps) {
  const isUser = message.role === 'user';

  return (
    <div className={`tw-flex ${isUser ? 'tw-justify-end' : 'tw-justify-start'} tw-mb-3`}>
      <div className={`tw-max-w-[80%] ...`}>
        {/* 新增：思考过程面板（仅 assistant） */}
        {!isUser && message.thinkingSteps && message.thinkingSteps.length > 0 && (
          <ThinkingPanel steps={message.thinkingSteps} />
        )}

        {/* 原有内容渲染 */}
        {message.content && (
          <span dangerouslySetInnerHTML={{ __html: renderMarkdown(message.content) }} />
        )}

        {/* 原有 loading 状态 */}
        {isStreaming && !message.content && (...)}
        {isStreaming && message.content && (...)}
      </div>
    </div>
  );
}
```

**具体修改点：**

| 行号范围 | 修改内容 |

| ~line 1（props） | `MessageBubbleProps` 的 `message` 类型改用带 `thinkingSteps` 的 `Message` |
| 新增 | `ThinkingPanel` 内联组件（或拆分为独立文件） |
| ~line 35（内容区） | 在 `renderMarkdown` 前插入 `<ThinkingPanel />` |

## 四、验收标准

1. 打开聊天，发送一个问题
2. 在最终回答出现前，能看到 🤔 → 🔍 → 📄 的逐步出现
3. 每步可点击折叠/展开
4. 回答完成后，思考面板默认折叠，只显示"🤔 思考过程（3 步）"
5. 切换历史会话，每条消息的思考过程都能正确恢复显示
